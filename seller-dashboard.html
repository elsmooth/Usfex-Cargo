// REPLACES EXISTING LOCALSTORAGE INGESTION ENGINE IN SELLER-DASHBOARD.HTML async function executeAssetVaultIngress(event) { event.preventDefault(); const payload = { title: document.getElementById('assetTitle').value, weight: document.getElementById('assetWeight').value, gross: document.getElementById('assetGross').value }; try { // Dispatching network request payload to our secure API layer const response = await fetch('http://localhost:5000/api/products/ingress', { method: 'POST', headers: { 'Content-Type': 'application/json', // Passes the user's secure token back to the verification gate 'Authorization': `Bearer ${localStorage.getItem('usfex_auth_token')}` }, body: JSON.stringify(payload) }); const data = await response.json(); if (response.ok) { alert(`ASSET MANIFEST DEPOSITED IN LIVE DB // SKU: ${data.asset.sku}\nSTATUS: AWAITING CLEARANCE.`); document.getElementById('ingressForm').reset(); window.location.href = 'index.html'; } else { alert(`[INGRESS_REJECTION] Error: ${data.error}`); } } catch (err) { console.error("API Communication Failure:", err); alert("[CRITICAL_NETWORK_FAULT] Could not establish connection to operations server."); } }
